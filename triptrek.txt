    <script type="module">
        // Import Firebase modules
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { 
            getAuth, 
            signInAnonymously, 
            signInWithCustomToken, 
            onAuthStateChanged,
            signOut
        } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { 
            getFirestore, 
            doc, 
            setDoc, 
            addDoc, 
            collection, 
            getDocs,
            onSnapshot,
            query,
            Timestamp,
            setLogLevel
        } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // --- Global Variables ---
        let db, auth, userId, isAuthReady = false;
        
        // Get app-specific config from the environment
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
        const firebaseConfig = JSON.parse(typeof __firebase_config !== 'undefined' ? __firebase_config : '{}');
        const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
        
        // --- Initialize Firebase ---
        try {
            const app = initializeApp(firebaseConfig);
            auth = getAuth(app);
            db = getFirestore(app);
            setLogLevel('Debug'); // Enable Firestore logging
            console.log("Firebase initialized.");
            initAuthentication();
        } catch (error) {
            console.error("Error initializing Firebase:", error);
            updateAuthStatusUI("Firebase init failed.");
        }

        /**
         * Handles user authentication state
         */
        function initAuthentication() {
            onAuthStateChanged(auth, user => {
                if (user) {
                    userId = user.uid;
                    isAuthReady = true;
                    console.log("User is signed in:", userId);
                    updateAuthStatusUI(`User: ${userId.substring(0, 8)}...`);
                    // Load dynamic content now that user is authenticated
                    loadHelpers();
                } else {
                    userId = null;
                    isAuthReady = false;
                    console.log("User is signed out.");
                    updateAuthStatusUI("Not signed in.");
                    // Attempt to sign in
                    signInUser();
                }
            });
        }
        
        /**
         * Tries to sign in with custom token or anonymously
         */
        async function signInUser() {
            try {
                if (initialAuthToken) {
                    console.log("Attempting sign in with custom token...");
                    await signInWithCustomToken(auth, initialAuthToken);
                } else {
                    console.log("Attempting anonymous sign in...");
                    await signInAnonymously(auth);
                }
            } catch (error) {
                console.error("Error signing in:", error);
                updateAuthStatusUI("Auth failed.");
            }
        }
        
        /**
         * Handles user logout
         */
        async function handleLogout() {
            try {
                await signOut(auth);
                // This will trigger onAuthStateChanged, which updates UI
                // But we'll try to sign in again anonymously
                console.log("User signed out.");
            } catch (error)
            {
                console.error("Error signing out:", error);
            }
        }

        /**
         * Updates all auth-related UI elements
         */
        function updateAuthStatusUI(statusText) {
            const authStatus = document.getElementById('auth-status');
            const mobileAuthStatus = document.getElementById('mobile-auth-status');
            const logoutBtn = document.getElementById('logout-btn');
            const mobileLogoutBtn = document.getElementById('mobile-logout-btn');

            if (authStatus) authStatus.innerText = statusText;
            if (mobileAuthStatus) mobileAuthStatus.innerText = statusText;

            if (isAuthReady) {
                if (logoutBtn) logoutBtn.classList.remove('hidden');
                if (mobileLogoutBtn) mobileLogoutBtn.classList.remove('hidden');
            } else {
                if (logoutBtn) logoutBtn.classList.add('hidden');
                if (mobileLogoutBtn) mobileLogoutBtn.classList.add('hidden');
            }
        }

         */

            }

        /**
         * Loads and displays helper profiles from Firestore in real-time
         */
        function loadHelpers() {
            if (!isAuthReady) return;
            
            const helperListDiv = document.getElementById('helper-list');
            const loadingP = document.getElementById('helpers-loading');
            const helpersCollectionRef = collection(db, `artifacts/${appId}/public/data/helpers`);
            
            const q = query(helpersCollectionRef);

            onSnapshot(q, (snapshot) => {
                if (snapshot.empty) {
                    if(loadingP) loadingP.innerText = "No local helpers found. Try seeding demo data!";
                    helperListDiv.innerHTML = ''; // Clear list
                } else {
                    if(loadingP) loadingP.style.display = 'none';
                    helperListDiv.innerHTML = ''; // Clear list before repopulating
                    
                    snapshot.forEach(doc => {
                        const helper = doc.data();
                        const helperId = doc.id;
                        const card = createHelperCard(helper, helperId);
                        helperListDiv.appendChild(card);
                    });
                    
                    // Re-run lucide to render new icons
                    lucide.createIcons();
                }
            }, (error) => {
                console.error("Error loading helpers:", error);
                if(loadingP) loadingP.innerText = "Error loading helpers.";
            });
        }
        
        /**
         * Creates the HTML element for a single helper card
         */
        function createHelperCard(helper, helperId) {
            const card = document.createElement('div');
            card.className = "bg-white p-6 rounded-lg shadow-lg border border-gray-200 flex flex-col";
            
            let specialtiesHTML = '';
            if (helper.specialties && helper.specialties.length > 0) {
                specialtiesHTML = helper.specialties.map(s => 
                    `<span class="bg-blue-100 text-blue-800 text-xs font-medium px-2.5 py-0.5 rounded-full">${s}</span>`
                ).join(' ');
            }

            card.innerHTML = `
                <img src="${helper.img}" alt="${helper.name}" class="w-full h-48 object-cover rounded-md mb-4" onerror="this.onerror=null;this.src='https://placehold.co/600x400/CCCCCC/FFFFFF?text=Helper';">
                <h4 class="text-2xl font-bold text-gray-800 mb-1">${helper.name}</h4>
                <div class="flex items-center text-gray-500 mb-3">
                    <i data-lucide="map-pin" class="w-4 h-4 mr-2"></i>
                    <span>${helper.location}</span>
                </div>
                <p class="text-gray-600 mb-4 flex-grow">${helper.bio}</p>
                <div class="flex flex-wrap gap-2 mb-6">
                    ${specialtiesHTML}
                </div>
                <button class="contact-helper-btn mt-auto w-full bg-blue-600 text-white px-5 py-3 rounded-lg font-semibold hover:bg-blue-700 transition-colors" data-helper-id="${helperId}" data-helper-name="${helper.name}">
                    Contact ${helper.name.split(' ')[0]}
                </button>
            `;
            return card;
        }

        /**
         * Handles the submission of the message form
         */
        async function handleSendMessage(e) {
            e.preventDefault();
            if (!isAuthReady || !userId) {
                showToast("You must be signed in to send a message.");
                return;
            }
            
            const helperId = document.getElementById('modal-helper-id').value;
            const messageText = document.getElementById('message-text').value;
            const sendBtn = document.getElementById('send-message-btn');

            sendBtn.disabled = true;
            sendBtn.innerText = 'Sending...';

            try {
                // Store message in a public collection for this app
                const messagesCollectionRef = collection(db, `artifacts/${appId}/public/data/messages`);
                await addDoc(messagesCollectionRef, {
                    senderId: userId,
                    receiverId: helperId,
                    message: messageText,
                    timestamp: Timestamp.now(),
                    read: false
                });
                
                showToast("Message sent successfully!");
                closeMessageModal();

            } catch (error) {
                console.error("Error sending message:", error);
                showToast("Error sending message. Please try again.");
            } finally {
                sendBtn.disabled = false;
                sendBtn.innerText = 'Send Message';
            }
        }

        /**
         * Fetches the itinerary from the Gemini API with exponential backoff.
         */
        async function fetchItinerary(destination, budget, duration, interests) {
            const systemPrompt = "You are TripTrek, an expert travel planning assistant. Your goal is to create a detailed, helpful, and inspiring travel itinerary. The user wants a plan for their trip. You MUST format your response in Markdown. Be enthusiastic and provide practical, budget-conscious tips where possible, with all cost estimates and budget considerations in Indian Rupees (INR). Start with a catchy title for the trip. Include a mix of activities, food suggestions, and local transport tips.";
            
            const userQuery = `Please create a ${duration}-day travel itinerary for ${destination}. 
My budget is ${budget} (in Indian Rupees - INR). 
My interests are: ${interests}.
Provide a day-by-day plan. For each day, suggest activities, food options (mentioning budget-friendly choices), and estimated local transport ideas (like using public transport, walking, etc.). 
Make the plan exciting and easy to follow.`;

            const apiKey = ""; // API key will be provided by the environment
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

            const payload = {
                contents: [{ parts: [{ text: userQuery }] }],
                systemInstruction: {
                    parts: [{ text: systemPrompt }]
                },
                generationConfig: {
                    temperature: 0.7,
                    topP: 0.95,
                    topK: 40,
                }
            };

            let attempt = 0;
            const maxAttempts = 3;
            let delay = 1000; // Start with 1 second

            while (attempt < maxAttempts) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }

                    const result = await response.json();
                    
                    if (result.candidates && result.candidates[0].content?.parts?.[0]?.text) {
                        return result.candidates[0].content.parts[0].text;
                    } else {
                        throw new Error("Invalid response structure from API.");
                    }

                } catch (error) {
                    console.warn(`Attempt ${attempt + 1} failed. Retrying in ${delay}ms...`, error);
                    attempt++;
                    if (attempt >= maxAttempts) {
                        throw new Error("Failed to generate itinerary after multiple attempts.");
                    }
                    await new Promise(resolve => setTimeout(resolve, delay));
                    delay *= 2; // Exponential backoff
                }
            }
        }
    </script>
